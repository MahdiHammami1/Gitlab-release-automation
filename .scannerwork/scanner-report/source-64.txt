import { HttpService } from '@nestjs/axios';
import { Test, TestingModule } from '@nestjs/testing';
import { of } from 'rxjs';
import { GitlabService } from './gitlab.service';

describe('GitlabService', () => {
  let service: GitlabService;
  let httpService: HttpService;

  beforeAll(() => {
    process.env.GITLAB_PAT = 'dummy_token';
  });

  beforeEach(async () => {
    const httpServiceMock = {
      get: jest.fn(),
      post: jest.fn(),
      put: jest.fn(),
      delete: jest.fn(),
      request: jest.fn(),
    };
    const module: TestingModule = await Test.createTestingModule({
      providers: [
        GitlabService,
        { provide: HttpService, useValue: httpServiceMock },
      ],
    }).compile();

    service = module.get<GitlabService>(GitlabService);
    httpService = module.get<HttpService>(HttpService);
  });

  it('should be defined', () => {
    expect(service).toBeDefined();
  });

  it('should return bearer token', () => {
    const result = service['bearer']('token123');
    expect(result).toEqual({ Authorization: 'Bearer token123' });
  });

  it('should return private token', () => {
    const result = service['privateToken']('token123');
    expect(result).toEqual({ 'PRIVATE-TOKEN': 'token123' });
  });

  // Suppression des tests inutilisables sur cleanParams et isNumericId

  it('should call get with bearer token', async () => {
    httpService.request = jest.fn().mockReturnValue(of({ data: 'ok' }));
    await expect(service.get('/projects')).resolves.toEqual('ok');
  });

  it('should call post with bearer token', async () => {
    httpService.request = jest.fn().mockReturnValue(of({ data: 'created' }));
    await expect(service.post('/projects', { name: 'test' })).resolves.toEqual('created');
  });

  it('should get user profile (me)', async () => {
    httpService.request = jest.fn().mockReturnValue(of({ data: { id: 1, name: 'user' } }));
    await expect(service.me()).resolves.toEqual({ id: 1, name: 'user' });
  });

  it('should handle error on me', async () => {
    httpService.request = jest.fn().mockImplementation(() => { throw new Error('HTTP error'); });
    await expect(service.me()).rejects.toThrow('HTTP error');
  });

  it('should get projects (myProjects)', async () => {
    httpService.request = jest.fn().mockReturnValue(of({ data: [{ id: 1, name: 'project' }] }));
    await expect(service.myProjects()).resolves.toEqual([{ id: 1, name: 'project' }]);
  });

  it('should handle error on myProjects', async () => {
    httpService.request = jest.fn().mockImplementation(() => { throw new Error('HTTP error'); });
    await expect(service.myProjects()).rejects.toThrow('HTTP error');
  });

  it('should get project by id', async () => {
    httpService.request = jest.fn().mockReturnValue(of({ data: { id: '1', name: 'project' } }));
    await expect(service.project('1')).resolves.toEqual({ id: '1', name: 'project' });
  });

  it('should return null if project by id not found', async () => {
    httpService.request = jest.fn().mockReturnValue(of({ data: null }));
    await expect(service.project('999')).resolves.toBeNull();
  });

  it('should handle error on project by id', async () => {
    httpService.request = jest.fn().mockImplementation(() => { throw new Error('HTTP error'); });
    await expect(service.project('1')).rejects.toThrow('HTTP error');
  });

  it('should get project by path', async () => {
    httpService.request = jest.fn().mockReturnValue(of({ data: { id: '1', name: 'project' } }));
    await expect(service.projectByPath('namespace/project')).resolves.toEqual({ id: '1', name: 'project' });
  });

  it('should return null if project by path not found', async () => {
    httpService.request = jest.fn().mockReturnValue(of({ data: null }));
    await expect(service.projectByPath('not/found')).resolves.toBeNull();
  });

  it('should handle error on project by path', async () => {
    httpService.request = jest.fn().mockImplementation(() => { throw new Error('HTTP error'); });
    await expect(service.projectByPath('namespace/project')).rejects.toThrow('HTTP error');
  });

  it('should get issues for a project', async () => {
    httpService.request = jest.fn().mockReturnValue(of({ data: [{ id: 1, title: 'issue' }] }));
    await expect(service.issues('1')).resolves.toEqual([{ id: 1, title: 'issue' }]);
  });

  it('should handle error on issues', async () => {
    httpService.request = jest.fn().mockImplementation(() => { throw new Error('HTTP error'); });
    await expect(service.issues('1')).rejects.toThrow('HTTP error');
  });

  it('should return empty array if issues not found', async () => {
    httpService.request = jest.fn().mockReturnValue(of({ data: [] }));
    await expect(service.issues('999')).resolves.toEqual([]);
  });
});
