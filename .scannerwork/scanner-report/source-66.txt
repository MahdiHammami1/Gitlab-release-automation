import { Test, TestingModule } from '@nestjs/testing';
import { ModuleReleasesController } from './module-releases.controller';
import { ModuleReleasesService } from './module-releases.service';

describe('ModuleReleasesController', () => {
  let controller: ModuleReleasesController;
  let serviceMock: any;

  beforeEach(async () => {
    serviceMock = {
      create: jest.fn(),
      findAll: jest.fn(),
      findOne: jest.fn(),
      update: jest.fn(),
      remove: jest.fn(),
    };
    const module: TestingModule = await Test.createTestingModule({
      controllers: [ModuleReleasesController],
      providers: [
        { provide: ModuleReleasesService, useValue: serviceMock },
      ],
    }).compile();

    controller = module.get<ModuleReleasesController>(ModuleReleasesController);
  });

  it('should be defined', () => {
    expect(controller).toBeDefined();
  });

  it('should create a module release if not exists', async () => {
    serviceMock.exists = jest.fn().mockResolvedValue(null);
    serviceMock.create = jest.fn().mockResolvedValue({ id: '1', name: 'release1', moduleId: 'm1', tagId: 't1' });
    const dto = { name: 'release1', moduleId: 'm1', tagId: 't1' };
    await expect(controller.create(dto)).resolves.toEqual({ id: '1', name: 'release1', moduleId: 'm1', tagId: 't1' });
    expect(serviceMock.exists).toHaveBeenCalledWith(dto);
    expect(serviceMock.create).toHaveBeenCalledWith(dto);
  });

  it('should return existing module release if exists', async () => {
    serviceMock.exists = jest.fn().mockResolvedValue({ id: '1', name: 'release1', moduleId: 'm1', tagId: 't1' });
    const dto = { name: 'release1', moduleId: 'm1', tagId: 't1' };
    await expect(controller.create(dto)).resolves.toEqual({ message: 'ModuleRelease déjà existant', moduleRelease: { id: '1', name: 'release1', moduleId: 'm1', tagId: 't1' } });
    expect(serviceMock.exists).toHaveBeenCalledWith(dto);
  });

  it('should find all module releases', async () => {
    serviceMock.findAll = jest.fn().mockResolvedValue([{ id: '1', name: 'release1' }]);
    await expect(controller.findAll()).resolves.toEqual([{ id: '1', name: 'release1' }]);
    expect(serviceMock.findAll).toHaveBeenCalled();
  });

  it('should find one module release', async () => {
    serviceMock.findOne = jest.fn().mockResolvedValue({ id: '1', name: 'release1' });
    await expect(controller.findOne('1')).resolves.toEqual({ id: '1', name: 'release1' });
    expect(serviceMock.findOne).toHaveBeenCalledWith('1');
  });

  it('should update a module release', async () => {
    serviceMock.update = jest.fn().mockResolvedValue({ id: '1', name: 'updated', moduleId: 'm1', tagId: 't1' });
    const dto = { name: 'updated', moduleId: 'm1', tagId: 't1' };
    await expect(controller.update('1', dto)).resolves.toEqual({ id: '1', name: 'updated', moduleId: 'm1', tagId: 't1' });
    expect(serviceMock.update).toHaveBeenCalledWith('1', dto);
  });

  it('should remove a module release', async () => {
    serviceMock.remove = jest.fn().mockResolvedValue({ id: '1', name: 'release1' });
    await expect(controller.remove('1')).resolves.toEqual({ id: '1', name: 'release1' });
    expect(serviceMock.remove).toHaveBeenCalledWith('1');
  });

  it('should handle error when creating a module release', async () => {
    serviceMock.exists = jest.fn().mockRejectedValue(new Error('Service error'));
    const dto = { name: 'release1', moduleId: 'm1', tagId: 't1' };
    await expect(controller.create(dto)).rejects.toThrow('Service error');
  });

  it('should handle error when finding all module releases', async () => {
    serviceMock.findAll = jest.fn().mockRejectedValue(new Error('Service error'));
    await expect(controller.findAll()).rejects.toThrow('Service error');
  });

  it('should handle error when finding one module release', async () => {
    serviceMock.findOne = jest.fn().mockRejectedValue(new Error('Service error'));
    await expect(controller.findOne('1')).rejects.toThrow('Service error');
  });

  it('should handle error when updating a module release', async () => {
    serviceMock.update = jest.fn().mockRejectedValue(new Error('Service error'));
    const dto = { name: 'updated', moduleId: 'm1', tagId: 't1' };
    await expect(controller.update('1', dto)).rejects.toThrow('Service error');
  });

  it('should handle error when removing a module release', async () => {
    serviceMock.remove = jest.fn().mockRejectedValue(new Error('Service error'));
    await expect(controller.remove('1')).rejects.toThrow('Service error');
  });

  it('should return null if module release not found', async () => {
    serviceMock.findOne = jest.fn().mockResolvedValue(null);
    await expect(controller.findOne('999')).resolves.toBeNull();
    expect(serviceMock.findOne).toHaveBeenCalledWith('999');
  });

  it('should return null if remove module release not found', async () => {
    serviceMock.remove = jest.fn().mockResolvedValue(null);
    await expect(controller.remove('999')).resolves.toBeNull();
    expect(serviceMock.remove).toHaveBeenCalledWith('999');
  });

  it('should return null if update module release not found', async () => {
    const dto = { name: 'notfound', moduleId: 'm1', tagId: 't1' };
    serviceMock.update = jest.fn().mockResolvedValue(null);
    await expect(controller.update('999', dto)).resolves.toBeNull();
    expect(serviceMock.update).toHaveBeenCalledWith('999', dto);
  });

  it('should handle invalid id for findOne', async () => {
    serviceMock.findOne = jest.fn().mockResolvedValue(null);
    await expect(controller.findOne('')).resolves.toBeNull();
    expect(serviceMock.findOne).toHaveBeenCalledWith('');
  });

  it('should handle invalid id for remove', async () => {
    serviceMock.remove = jest.fn().mockResolvedValue(null);
    await expect(controller.remove('')).resolves.toBeNull();
    expect(serviceMock.remove).toHaveBeenCalledWith('');
  });
});
