import { HttpService } from '@nestjs/axios';
import { Test, TestingModule } from '@nestjs/testing';
import { ModulesService } from './modules.service';
import { PrismaService } from '../prisma/prisma.service';
import { of } from 'rxjs';

describe('ModulesService', () => {
  let service: ModulesService;
  let prisma: PrismaService;
  let httpService: HttpService;

  beforeEach(async () => {
    process.removeAllListeners('beforeExit');
    jest.clearAllMocks();
    const prismaMock = {
      module: {
        create: jest.fn(),
        findMany: jest.fn(),
        findUnique: jest.fn(),
        update: jest.fn(),
        delete: jest.fn(),
      },
    };
    const httpServiceMock = {
      get: jest.fn(),
      post: jest.fn(),
      put: jest.fn(),
      delete: jest.fn(),
      request: jest.fn(),
    };
    const module: TestingModule = await Test.createTestingModule({
      providers: [
        ModulesService,
        { provide: PrismaService, useValue: prismaMock },
        { provide: HttpService, useValue: httpServiceMock },
      ],
    }).compile();

    service = module.get<ModulesService>(ModulesService);
    prisma = module.get<PrismaService>(PrismaService);
    httpService = module.get<HttpService>(HttpService);
  });

  afterEach(() => {
    process.removeAllListeners('beforeExit');
    jest.clearAllMocks();
  });

  it('should be defined', () => {
    expect(service).toBeDefined();
  });

  it('should create a module', async () => {
    const dto = { name: 'module1', repoUrl: 'https://repo.url' };
    prisma.module.create = jest.fn().mockResolvedValue({ id: '1', ...dto });
    await expect(service.create(dto)).resolves.toEqual({ id: '1', ...dto });
    expect(prisma.module.create).toHaveBeenCalledWith({ data: dto });
  });

  it('should find all modules', async () => {
    prisma.module.findMany = jest.fn().mockResolvedValue([{ id: '1', name: 'module1' }]);
    await expect(service.findAll()).resolves.toEqual([{ id: '1', name: 'module1' }]);
    expect(prisma.module.findMany).toHaveBeenCalled();
  });

  it('should find one module', async () => {
    prisma.module.findUnique = jest.fn().mockResolvedValue({ id: '1', name: 'module1' });
    await expect(service.findOne('1')).resolves.toEqual({ id: '1', name: 'module1' });
    expect(prisma.module.findUnique).toHaveBeenCalledWith({ where: { id: '1' } });
  });

  it('should update a module', async () => {
    const dto = { name: 'updated' };
    prisma.module.update = jest.fn().mockResolvedValue({ id: '1', ...dto });
    await expect(service.update('1', dto as any)).resolves.toEqual({ id: '1', ...dto });
    expect(prisma.module.update).toHaveBeenCalledWith({ where: { id: '1' }, data: dto });
  });

  it('should remove a module', async () => {
    prisma.module.delete = jest.fn().mockResolvedValue({ id: '1', name: 'module1' });
    await expect(service.remove('1')).resolves.toEqual({ id: '1', name: 'module1' });
    expect(prisma.module.delete).toHaveBeenCalledWith({ where: { id: '1' } });
  });

  it('should handle prisma error on create', async () => {
    const dto = { name: 'fail', repoUrl: 'https://repo.url' };
    prisma.module.create = jest.fn().mockRejectedValue(new Error('Prisma error'));
    await expect(service.create(dto)).rejects.toThrow('Prisma error');
  });

  it('should handle prisma error on update', async () => {
    const dto = { name: 'fail', repoUrl: 'https://repo.url' };
    prisma.module.update = jest.fn().mockRejectedValue(new Error('Prisma error'));
    await expect(service.update('1', dto)).rejects.toThrow('Prisma error');
  });

  it('should handle prisma error on remove', async () => {
    prisma.module.delete = jest.fn().mockRejectedValue(new Error('Prisma error'));
    await expect(service.remove('1')).rejects.toThrow('Prisma error');
  });
});
