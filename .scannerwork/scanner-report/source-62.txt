import { Test, TestingModule } from '@nestjs/testing';
import { GitlabController } from './gitlab.controller';
import { GitlabService } from './gitlab.service';
import { HttpService } from '@nestjs/axios';

describe('GitlabController', () => {
  let controller: GitlabController;
  let serviceMock: any;

  beforeEach(async () => {
    serviceMock = {
      getProject: jest.fn(),
      getUser: jest.fn(),
      createProject: jest.fn(),
      // Ajoutez ici les autres méthodes du service à mocker si besoin
    };
    const httpServiceMock = {
      get: jest.fn(),
      post: jest.fn(),
      put: jest.fn(),
      delete: jest.fn(),
      request: jest.fn(),
    };
    const module: TestingModule = await Test.createTestingModule({
      controllers: [GitlabController],
      providers: [
        { provide: GitlabService, useValue: serviceMock },
        { provide: HttpService, useValue: httpServiceMock },
      ],
    }).compile();

    controller = module.get<GitlabController>(GitlabController);
  });

  it('should be defined', () => {
    expect(controller).toBeDefined();
  });

  it('should return debug info', () => {
    const result = controller.debug();
    expect(result).toHaveProperty('api');
    expect(result).toHaveProperty('pat_masked');
  });

  it('should return user profile', async () => {
    serviceMock.me = jest.fn().mockResolvedValue({ id: 1, name: 'user' });
    await expect(controller.me()).resolves.toEqual({ id: 1, name: 'user' });
    expect(serviceMock.me).toHaveBeenCalled();
  });

  it('should return projects', async () => {
    serviceMock.myProjects = jest.fn().mockResolvedValue([{ id: 1, name: 'project' }]);
    await expect(controller.projects('search', 'true', 'public', 1, 50, 'last_activity_at', 'desc')).resolves.toEqual([{ id: 1, name: 'project' }]);
    expect(serviceMock.myProjects).toHaveBeenCalled();
  });

  it('should return project by id', async () => {
    serviceMock.project = jest.fn().mockResolvedValue({ id: '1', name: 'project' });
    await expect(controller.projectById('1')).resolves.toEqual({ id: '1', name: 'project' });
    expect(serviceMock.project).toHaveBeenCalledWith('1');
  });

  it('should return project by path', async () => {
    serviceMock.projectByPath = jest.fn().mockResolvedValue({ id: '1', name: 'project' });
    await expect(controller.projectByPath('namespace/project')).resolves.toEqual({ id: '1', name: 'project' });
    expect(serviceMock.projectByPath).toHaveBeenCalledWith('namespace/project');
  });

  it('should return issues for a project', async () => {
    serviceMock.issues = jest.fn().mockResolvedValue([{ id: 1, title: 'issue' }]);
    await expect(controller.issues('1', 'all', 1)).resolves.toEqual([{ id: 1, title: 'issue' }]);
    expect(serviceMock.issues).toHaveBeenCalledWith('1', { state: 'all', page: 1, per_page: 50 });
  });

  it('should handle error on me', async () => {
    serviceMock.me = jest.fn().mockRejectedValue(new Error('Service error'));
    await expect(controller.me()).rejects.toThrow('Service error');
  });

  it('should handle error on projects', async () => {
    serviceMock.myProjects = jest.fn().mockRejectedValue(new Error('Service error'));
    await expect(controller.projects('search', 'true', 'public', 1, 50, 'last_activity_at', 'desc')).rejects.toThrow('Service error');
  });

  it('should handle error on projectById', async () => {
    serviceMock.project = jest.fn().mockRejectedValue(new Error('Service error'));
    await expect(controller.projectById('1')).rejects.toThrow('Service error');
  });

  it('should handle error on projectByPath', async () => {
    serviceMock.projectByPath = jest.fn().mockRejectedValue(new Error('Service error'));
    await expect(controller.projectByPath('namespace/project')).rejects.toThrow('Service error');
  });

  it('should handle error on issues', async () => {
    serviceMock.issues = jest.fn().mockRejectedValue(new Error('Service error'));
    await expect(controller.issues('1', 'all', 1)).rejects.toThrow('Service error');
  });

  it('should return null if projectById not found', async () => {
    serviceMock.project = jest.fn().mockResolvedValue(null);
    await expect(controller.projectById('999')).resolves.toBeNull();
  });

  it('should return null if projectByPath not found', async () => {
    serviceMock.projectByPath = jest.fn().mockResolvedValue(null);
    await expect(controller.projectByPath('not/found')).resolves.toBeNull();
  });

  it('should handle invalid id for projectById', async () => {
    serviceMock.project = jest.fn().mockResolvedValue(null);
    await expect(controller.projectById('')).resolves.toBeNull();
  });

  it('should handle invalid id for issues', async () => {
    serviceMock.issues = jest.fn().mockResolvedValue([]);
    await expect(controller.issues('', 'all', 1)).resolves.toEqual([]);
  });

  // Ajouter ici d'autres tests pour chaque endpoint du contrôleur
});
