import { Test, TestingModule } from '@nestjs/testing';
import { ModuleReleasesService } from './module-releases.service';
import { PrismaService } from '../prisma/prisma.service';
import { HttpException } from '@nestjs/common';

describe('ModuleReleasesService', () => {
  let service: ModuleReleasesService;
  let prisma: PrismaService;

  beforeEach(async () => {
    const prismaMock = {
      moduleRelease: {
        create: jest.fn(),
        findMany: jest.fn(),
        findUnique: jest.fn(),
        update: jest.fn(),
        delete: jest.fn(),
      },
    };
    const module: TestingModule = await Test.createTestingModule({
      providers: [
        ModuleReleasesService,
        { provide: PrismaService, useValue: prismaMock },
      ],
    }).compile();

    service = module.get<ModuleReleasesService>(ModuleReleasesService);
    prisma = module.get<PrismaService>(PrismaService);
  });

  it('should be defined', () => {
    expect(service).toBeDefined();
  });

  it('should create a module release', async () => {
    const dto = { name: 'release1' } as any;
    prisma.moduleRelease.create = jest.fn().mockResolvedValue({ id: '1', ...dto });
    await expect(service.create(dto)).resolves.toEqual({ id: '1', ...dto });
    expect(prisma.moduleRelease.create).toHaveBeenCalledWith({ data: dto });
  });

  it('should find all module releases', async () => {
    prisma.moduleRelease.findMany = jest.fn().mockResolvedValue([{ id: '1', name: 'release1' }]);
    await expect(service.findAll()).resolves.toEqual([{ id: '1', name: 'release1' }]);
    expect(prisma.moduleRelease.findMany).toHaveBeenCalledWith({
      include: {
        module: true,
        tag: true,
        artefacts: true,
        release: true,
      },
    });
  });

  it('should find one module release', async () => {
    prisma.moduleRelease.findUnique = jest.fn().mockResolvedValue({ id: '1', name: 'release1' });
    await expect(service.findOne('1')).resolves.toEqual({ id: '1', name: 'release1' });
    expect(prisma.moduleRelease.findUnique).toHaveBeenCalledWith({
      where: { id: '1' },
      include: {
        module: true,
        tag: true,
        artefacts: true,
        release: true,
      },
    });
  });

  it('should update a module release', async () => {
    const dto = { name: 'updated' } as any;
    prisma.moduleRelease.update = jest.fn().mockResolvedValue({ id: '1', ...dto });
    await expect(service.update('1', dto)).resolves.toEqual({ id: '1', ...dto });
    expect(prisma.moduleRelease.update).toHaveBeenCalledWith({ where: { id: '1' }, data: dto });
  });

  it('should remove a module release', async () => {
    prisma.moduleRelease.delete = jest.fn().mockResolvedValue({ id: '1', name: 'release1' });
    await expect(service.remove('1')).resolves.toEqual({ id: '1', name: 'release1' });
    expect(prisma.moduleRelease.delete).toHaveBeenCalledWith({ where: { id: '1' } });
  });

  it('should check if module release exists', async () => {
    prisma.moduleRelease.findFirst = jest.fn().mockResolvedValue({ id: '1', name: 'release1' });
    await expect(service.exists({ moduleId: 'm1', tagId: 't1', releaseId: 'r1' })).resolves.toEqual({ id: '1', name: 'release1' });
    expect(prisma.moduleRelease.findFirst).toHaveBeenCalledWith({ where: { moduleId: 'm1', tagId: 't1', releaseId: 'r1' } });
  });

  it('should return null if module release does not exist', async () => {
    prisma.moduleRelease.findFirst = jest.fn().mockResolvedValue(null);
    await expect(service.exists({ moduleId: 'm1', tagId: 't1', releaseId: 'r1' })).resolves.toBeNull();
  });

  it('should handle error on create', async () => {
    const dto = { name: 'fail' } as any;
    prisma.moduleRelease.create = jest.fn().mockRejectedValue(new Error('DB error'));
    await expect(service.create(dto)).rejects.toThrow('DB error');
  });

  it('should handle error on findAll', async () => {
    prisma.moduleRelease.findMany = jest.fn().mockRejectedValue(new Error('DB error'));
    await expect(service.findAll()).rejects.toThrow('DB error');
  });

  it('should handle error on findOne', async () => {
    prisma.moduleRelease.findUnique = jest.fn().mockRejectedValue(new Error('DB error'));
    await expect(service.findOne('fail')).rejects.toThrow('DB error');
  });

  it('should return null if findOne not found', async () => {
    prisma.moduleRelease.findUnique = jest.fn().mockResolvedValue(null);
    await expect(service.findOne('999')).resolves.toBeNull();
  });

  it('should handle error on update', async () => {
    const dto = { name: 'fail' } as any;
    prisma.moduleRelease.update = jest.fn().mockRejectedValue(new Error('DB error'));
    await expect(service.update('999', dto)).rejects.toThrow('DB error');
  });

  it('should return null if update not found', async () => {
    const dto = { name: 'notfound' } as any;
    prisma.moduleRelease.update = jest.fn().mockResolvedValue(null);
    await expect(service.update('999', dto)).resolves.toBeNull();
  });

  it('should handle error on remove', async () => {
    prisma.moduleRelease.delete = jest.fn().mockRejectedValue(new Error('DB error'));
    await expect(service.remove('999')).rejects.toThrow('DB error');
  });

  it('should return null if remove not found', async () => {
    prisma.moduleRelease.delete = jest.fn().mockResolvedValue(null);
    await expect(service.remove('999')).resolves.toBeNull();
  });

  it('should handle error on exists', async () => {
    prisma.moduleRelease.findFirst = jest.fn().mockRejectedValue(new Error('DB error'));
    await expect(service.exists({ moduleId: 'm1', tagId: 't1', releaseId: 'r1' })).rejects.toThrow('DB error');
  });

  it('should throw HttpException if DB fails on findAll', async () => {
    prisma.moduleRelease.findMany = jest.fn().mockImplementation(() => { throw new Error('DB error'); });
    await expect(service.findAll()).rejects.toThrow(HttpException);
    await expect(service.findAll()).rejects.toThrow('Failed to retrieve releases');
  });
});
