import { Test, TestingModule } from '@nestjs/testing';
import { ReleasesController } from './releases.controller';
import { ReleasesService } from './releases.service';

describe('ReleasesController', () => {
  let controller: ReleasesController;
  let serviceMock: any;

  beforeEach(async () => {
    serviceMock = {
      create: jest.fn(),
      findAll: jest.fn(),
      findOne: jest.fn(),
      update: jest.fn(),
      remove: jest.fn(),
    };
    const module: TestingModule = await Test.createTestingModule({
      controllers: [ReleasesController],
      providers: [
        { provide: ReleasesService, useValue: serviceMock },
      ],
    }).compile();

    controller = module.get<ReleasesController>(ReleasesController);
  });

  it('should be defined', () => {
    expect(controller).toBeDefined();
  });

  it('should create a release', async () => {
    const dto = { name: 'release1', author: 'author1', changelogGlobal: 'changelog' };
    serviceMock.create.mockResolvedValue({ id: '1', ...dto });
    await expect(controller.create(dto)).resolves.toEqual({ id: '1', ...dto });
    expect(serviceMock.create).toHaveBeenCalledWith(dto);
  });

  it('should handle error on create', async () => {
    const dto = { name: 'release1', author: 'author1', changelogGlobal: 'changelog' };
    serviceMock.create.mockRejectedValue(new Error('Service error'));
    await expect(controller.create(dto)).rejects.toThrow('Service error');
  });

  it('should find all releases', async () => {
    serviceMock.findAll.mockResolvedValue([{ id: '1', name: 'release1' }]);
    await expect(controller.findAll()).resolves.toEqual([{ id: '1', name: 'release1' }]);
    expect(serviceMock.findAll).toHaveBeenCalled();
  });

  it('should handle error on findAll', async () => {
    serviceMock.findAll.mockRejectedValue(new Error('Service error'));
    await expect(controller.findAll()).rejects.toThrow('Service error');
  });

  it('should find one release', async () => {
    serviceMock.findOne.mockResolvedValue({ id: '1', name: 'release1' });
    await expect(controller.findOne('1')).resolves.toEqual({ id: '1', name: 'release1' });
    expect(serviceMock.findOne).toHaveBeenCalledWith('1');
  });

  it('should return null if release not found', async () => {
    serviceMock.findOne.mockResolvedValue(null);
    await expect(controller.findOne('999')).resolves.toBeNull();
    expect(serviceMock.findOne).toHaveBeenCalledWith('999');
  });

  it('should handle error on findOne', async () => {
    serviceMock.findOne.mockRejectedValue(new Error('Service error'));
    await expect(controller.findOne('1')).rejects.toThrow('Service error');
  });

  it('should update a release', async () => {
    const dto = { name: 'updated' };
    serviceMock.update.mockResolvedValue({ id: '1', ...dto });
    await expect(controller.update('1', dto)).resolves.toEqual({ id: '1', ...dto });
    expect(serviceMock.update).toHaveBeenCalledWith('1', dto);
  });

  it('should return null if update release not found', async () => {
    const dto = { name: 'notfound' };
    serviceMock.update.mockResolvedValue(null);
    await expect(controller.update('999', dto)).resolves.toBeNull();
    expect(serviceMock.update).toHaveBeenCalledWith('999', dto);
  });

  it('should handle error on update', async () => {
    const dto = { name: 'updated' };
    serviceMock.update.mockRejectedValue(new Error('Service error'));
    await expect(controller.update('1', dto)).rejects.toThrow('Service error');
  });

  it('should remove a release', async () => {
    serviceMock.remove.mockResolvedValue({ id: '1', name: 'deleted' });
    await expect(controller.remove('1')).resolves.toEqual({ id: '1', name: 'deleted' });
    expect(serviceMock.remove).toHaveBeenCalledWith('1');
  });

  it('should return null if remove release not found', async () => {
    serviceMock.remove.mockResolvedValue(null);
    await expect(controller.remove('999')).resolves.toBeNull();
    expect(serviceMock.remove).toHaveBeenCalledWith('999');
  });

  it('should handle error on remove', async () => {
    serviceMock.remove.mockRejectedValue(new Error('Service error'));
    await expect(controller.remove('1')).rejects.toThrow('Service error');
  });

  it('should handle invalid id for findOne', async () => {
    serviceMock.findOne.mockResolvedValue(null);
    await expect(controller.findOne('')).resolves.toBeNull();
    expect(serviceMock.findOne).toHaveBeenCalledWith('');
  });

  it('should handle invalid id for remove', async () => {
    serviceMock.remove.mockResolvedValue(null);
    await expect(controller.remove('')).resolves.toBeNull();
    expect(serviceMock.remove).toHaveBeenCalledWith('');
  });
});
